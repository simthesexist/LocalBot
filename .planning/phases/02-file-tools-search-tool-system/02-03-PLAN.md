# Plan 02-03 — code_search via @vscode/ripgrep + UI polish + final smoke

**As a** Localbot user,
**I want to** ask the bot to ripgrep-search the workspace with a regex and a glob, see the matching lines render inline, and cancel a long search mid-flight,
**so that** the Phase 2 file-tool surface is complete and the renderer handles long tool outputs cleanly.

## Objective

Close out Phase 2 with three additions:

1. **`code_search` tool** — replaces the Wave 1 stub with a real ripgrep implementation (binary resolution via `@vscode/ripgrep`, regex + glob, max_results cap, 60s timeout, audit-log payload minimization).
2. **Renderer polish** — collapse `tool_use` JSON + `tool_result` content over 500 bytes with a `Show more` toggle; error-tinted blocks; empty/single-line blocks render without the collapse UI.
3. **Final smoke (`smoke-tools.test.ts`)** — full headed Electron run gated by `LOCALBOT_SMOKE_OK`: fake M3 emits a `tool_use {name:'read_file', input:{path:'hello.txt'}}`, daemon executes, renderer shows BOTH the `tool_use` AND `tool_result` blocks inline.

## Purpose

`code_search` introduces a new subprocess surface (ripgrep child process). The risk is two-fold: (a) `tools/cancel` must actually kill the ripgrep process (Pitfall 3 in RESEARCH.md), and (b) a large result set would balloon the audit JSONL line above the 1 MiB cap (Pitfall 5). The plan addresses both with explicit child-tracking in the registry and a payload-minimizing audit shape. The renderer polish is the UI-03 finishing touch — without collapse, a 10 000-match code_search result would dump the entire chat pane.

## Output

- `code_search` tool that integrates with `@vscode/ripgrep@1.18.0` (already in package.json from Wave 1).
- `tools/cancel` kills the in-flight ripgrep child within 2s.
- `tool_use` + `tool_result` blocks render with collapse-for-length, error-tinted background for failures.
- 1 new Vitest suite (code_search) + 1 new Playwright headed smoke (smoke-tools.test.ts) + extended fake-m3-server.ts.

### A. Implement `daemon/tools/code_search.cjs`

Replaces the Wave 1 stub. Uses `@vscode/ripgrep` for binary resolution, spawns with explicit stdio:

```javascript
// daemon/tools/code_search.cjs
const path = require('node:path');
const fs = require('node:fs');
const { spawn } = require('node:child_process');
const readline = require('node:readline');
const { safePath } = require('./safe_path.cjs');

const MAX_RESULTS_DEFAULT = 200;
const TIMEOUT_MS = 60_000;
const RG_PATH = resolveRipgrepBinary();

function resolveRipgrepBinary() {
  try {
    const pkgPath = require.resolve('@vscode/ripgrep/package.json');
    const bin = process.platform === 'win32' ? 'rg.exe' : 'rg';
    return path.join(path.dirname(pkgPath), 'bin', bin);
  } catch {
    return null;
  }
}

function err(code, message) {
  const e = new Error(message);
  e.code = code;
  return e;
}

function stripLeadingDotSlash(p) {
  return p.replace(/^\.\//, '');
}

async function call({ pattern, glob, path: searchPath, max_results = MAX_RESULTS_DEFAULT }, ctx) {
  if (typeof pattern !== 'string' || pattern.length === 0) {
    throw err('invalid_pattern', 'pattern must be a non-empty string');
  }
  if (pattern.length > 1024 * 1024) {
    throw err('pattern_too_long', 'pattern exceeds 1 MiB');
  }
  if (pattern.includes('\0')) {
    throw err('invalid_pattern', 'pattern contains NUL byte');
  }
  if (glob != null) {
    if (typeof glob !== 'string' || glob.length === 0) {
      throw err('invalid_glob', 'glob must be a non-empty string');
    }
    if (glob.includes('\0')) {
      throw err('invalid_glob', 'glob contains NUL byte');
    }
  }
  const requested = typeof searchPath === 'string' && searchPath.length > 0 ? searchPath : '.';

  if (!RG_PATH || !fs.existsSync(RG_PATH)) {
    throw err('rg_not_found', 'ripgrep binary not found; run npm install');
  }

  const resolvedPath = await safePath(ctx.workspaceRoot, requested);

  const args = ['--json', '--no-messages', '--no-config', '--regexp', pattern];
  if (glob) args.push('--glob', glob);
  args.push(resolvedPath);

  const child = spawn(RG_PATH, args, {
    cwd: ctx.workspaceRoot,
    stdio: ['ignore', 'pipe', 'pipe'],
    // NEVER `shell: true` — args must be passed as array.
  });

  // Register the child in the registry so tools/cancel can kill it.
  if (ctx.toolCallId && ctx.registry && typeof ctx.registry.registerChild === 'function') {
    ctx.registry.registerChild(ctx.toolCallId, child);
  }

  const matches = [];
  let truncated = false;
  let linesSearched = 0;
  let stderrBuf = '';

  const rl = readline.createInterface({ input: child.stdout });
  const onAbort = () => {
    try { child.kill('SIGTERM'); } catch {}
    setTimeout(() => { try { child.kill('SIGKILL'); } catch {} }, 2000);
  };
  if (ctx.signal) ctx.signal.addEventListener('abort', onAbort);
  child.stderr.on('data', (b) => { stderrBuf += b.toString('utf8'); });

  const timeout = setTimeout(() => {
    try { child.kill('SIGTERM'); } catch {}
    setTimeout(() => { try { child.kill('SIGKILL'); } catch {} }, 2000);
  }, TIMEOUT_MS);

  try {
    for await (const line of rl) {
      if (!line) continue;
      let evt;
      try { evt = JSON.parse(line); } catch { continue; }
      if (evt.type === 'match' && evt.data) {
        const d = evt.data;
        const m = {
          path: stripLeadingDotSlash(d.path?.text ?? ''),
          line: d.line_number ?? 0,
          text: (d.lines?.text ?? '').replace(/\n$/, ''),
          submatches: (d.submatches ?? []).map((s) => ({
            text: s.match?.text ?? '',
            start: s.start ?? 0,
            end: s.end ?? 0,
          })),
        };
        if (matches.length < max_results) {
          matches.push(m);
        } else {
          truncated = true;
          break;
        }
      } else if (evt.type === 'context' || evt.type === 'begin') {
        // begin/context events: ignore for the result count; track only.
      } else if (evt.type === 'end' && evt.data?.stats?.lines_searched != null) {
        linesSearched = evt.data.stats.lines_searched;
      }
    }
    const code = await new Promise((resolve) => child.on('exit', resolve));
    if (code !== 0 && matches.length === 0 && !ctx.signal?.aborted) {
      // ripgrep returns 1 if no matches found (not an error); 2+ is a real error.
      if (code === 1) {
        // No matches — return empty result.
      } else if (code >= 2) {
        throw err('rg_error', `ripgrep exit ${code}: ${stderrBuf.slice(0, 500)}`);
      }
    }
  } finally {
    clearTimeout(timeout);
    if (ctx.signal) ctx.signal.removeEventListener('abort', onAbort);
    if (ctx.toolCallId && ctx.registry && typeof ctx.registry.unregisterChild === 'function') {
      ctx.registry.unregisterChild(ctx.toolCallId);
    }
  }

  return {
    matches,
    truncated,
    stats: { matches: matches.length, lines_searched: linesSearched },
  };
}

module.exports = { call };
```

// Additions to registry.cjs

const activeChildren = new Map(); // toolCallId -> ChildProcess

function registerChild(toolCallId, child) {
  activeChildren.set(toolCallId, child);
  child.once('exit', () => {
    if (activeChildren.get(toolCallId) === child) activeChildren.delete(toolCallId);
  });
}

function unregisterChild(toolCallId) {
  activeChildren.delete(toolCallId);
}

function cancelToolCall(toolCallId) {
  const child = activeChildren.get(toolCallId);
  if (!child) return { cancelled: false, reason: 'not_found' };
  try { child.kill('SIGTERM'); } catch {}
  setTimeout(() => {
    if (activeChildren.get(toolCallId) === child) {
      try { child.kill('SIGKILL'); } catch {}
    }
  }, 2000);
  return { cancelled: true };
}

// Updated callTool signature: callTool(botId, name, args, ctx)
// ctx = {workspaceRoot, toolCallId, signal, registry}
async function callTool(botId, name, args, ctx) {
  // ... existing allowlist checks ...

  return await mod.call(args, {
    ...ctx,
    registry: { registerChild, unregisterChild },
  });
}

```typescript
// src/renderer/components/ToolResultBlock.tsx
import { useState } from 'react';

const COLLAPSE_THRESHOLD_BYTES = 500;

function utf8ByteLength(s: string): number {
  // Browser-safe: TextEncoder is available in all modern renderers.
  return new TextEncoder().encode(s).length;
}

export interface ToolResultBlockProps {
  content: string;
  isError: boolean;
}

export function ToolResultBlock({ content, isError }: ToolResultBlockProps) {
  const [expanded, setExpanded] = useState(false);
  const byteLen = utf8ByteLength(content);
  const shouldCollapse = byteLen > COLLAPSE_THRESHOLD_BYTES;

  let display = content;
  if (shouldCollapse && !expanded) {
    // Slice on the first 500 bytes — find the largest char index whose
    // UTF-8 prefix length is <= 500.
    let i = 0;
    let bytes = 0;
    while (i < content.length && bytes <= COLLAPSE_THRESHOLD_BYTES) {
      // Approximation: JS strings are UTF-16; we still count by encoding each char.
      // For the threshold's purpose, a per-char iteration is sufficient (the
      // result is always within ±2 bytes of the target due to multi-byte chars).
      const charBytes = new TextEncoder().encode(content[i]).length;
      if (bytes + charBytes > COLLAPSE_THRESHOLD_BYTES) break;
      bytes += charBytes;
      i++;
    }
    display = content.slice(0, i);
  }

  const cls = `block-tool-result ${isError ? 'block-result-error' : ''}`;
  if (content.length === 0) {
    return <div className={cls}><span className="block-result-empty">(no output)</span></div>;
  }
  return (
    <div className={cls}>
      <pre className="block-result-content">{display}</pre>
      {shouldCollapse && (
        <button
          type="button"
          className="collapse-toggle"
          onClick={() => setExpanded((v) => !v)}
          data-testid={isError ? 'collapse-error' : 'collapse-result'}
        >
          {expanded ? 'Show less' : `Show more (+${byteLen - COLLAPSE_THRESHOLD_BYTES} bytes)`}
        </button>
      )}
    </div>
  );
}
```

// tests/playwright/fake-m3-server.ts

function streamToolUseResponse(
  res: http.ServerResponse,
  opts: { name: string; input: Record<string, unknown>; followupText: string },
): Promise<void> {
  const toolUseId = `toolu_test_${Date.now()}`;
  const inputJson = JSON.stringify(opts.input);
  const inputChunks: string[] = [];
  const CHUNK_SIZE = 12;
  for (let i = 0; i < inputJson.length; i += CHUNK_SIZE) {
    inputChunks.push(inputJson.slice(i, i + CHUNK_SIZE));
  }
  const tokens = opts.followupText.split(/(\s+)/).filter(Boolean);

  const chunks: string[] = [];
  chunks.push(
    sseFrame('message_start', {
      type: 'message_start',
      message: {
        id: 'msg_fake_tool',
        type: 'message',
        role: 'assistant',
        content: [],
        model: 'MiniMax/M3',
        stop_reason: null,
        stop_sequence: null,
        usage: { input_tokens: 5, output_tokens: 0 },
      },
    }),
    sseFrame('content_block_start', {
      type: 'content_block_start',
      index: 0,
      content_block: { type: 'tool_use', id: toolUseId, name: opts.name, input: {} },
    }),
  );
  inputChunks.forEach((chunk) => {
    chunks.push(
      sseFrame('content_block_delta', {
        type: 'content_block_delta',
        index: 0,
        delta: { type: 'input_json_delta', partial_json: chunk },
      }),
    );
  });
  chunks.push(
    sseFrame('content_block_stop', { type: 'content_block_stop', index: 0 }),
    sseFrame('content_block_start', {
      type: 'content_block_start',
      index: 1,
      content_block: { type: 'text', text: '' },
    }),
  );
  tokens.forEach((tok) => {
    chunks.push(
      sseFrame('content_block_delta', {
        type: 'content_block_delta',
        index: 1,
        delta: { type: 'text_delta', text: tok },
      }),
    );
  });
  chunks.push(
    sseFrame('content_block_stop', { type: 'content_block_stop', index: 1 }),
    sseFrame('message_delta', { type: 'message_delta', delta: { stop_reason: 'tool_use', stop_sequence: null } }),
